184template<typename Type>
185aJsonObject * getCreateObject(aJsonObject * a, int n, Type val)
186{
187if (!a) return NULL;
188if (a->type == aJson_Array)
189 {
190 aJsonObject * element = aJson.getArrayItem(a, n);
191 if (!element)
192 {
193 for (int i = aJson.getArraySize(a); i < n; i++)
194 if (i==n-1)
195 aJson.addItemToArray(a, element = aJson.createItem(val));
196 else aJson.addItemToArray(a, element = aJson.createNull());
197 }
198 return element;
199 }
200 return NULL;
201}

205 template<typename Type>
206 aJsonObject * getCreateObject(aJsonObject * a, const char * name, Type def)
207{
208if (!a) return NULL;
209if (a->type == aJson_Object)
210 {
211 aJsonObject * element = aJson.getObjectItem(a, name);
212 if (!element)
213 {
214 aJson.addNumberToObject(a, name, def);
215 element = aJson.getObjectItem(a, name);
216 }
217 return element;
218 }
219return NULL;
220}

222 template<typename Type>
223 aJsonObject * getCreateObject(aJsonObject * a, const char * name, int n, Type def)
224{
225if (!a) return NULL;
226if (a->type == aJson_Object) return getCreateObject(a,name,def);
227else if (a->type == aJson_Array) return getCreateObject(a,n,def);
228else return NULL;
229}

231 template<typename Type>
232 aJsonObject * setValToJson(aJsonObject * a, const char * name, Type val)
233{
234aJsonObject * element = getCreateObject(a,name);
235if (element)
236 switch (element->type)
237 {
238 case aJson_Float: element->valuefloat = val;
239 break;
240 case aJson_NULL: element->type = aJson_Int;
241 case aJson_Int: element->valueint = val;
242 }
243return element;
244}
